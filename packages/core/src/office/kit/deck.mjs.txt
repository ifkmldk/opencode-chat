// OpenCode office kit: slide decks with pptxgenjs. A small layout system (12-column grid, one accent, left-aligned
// type, generous margins) so decks look composed instead of generated. Compose slides with the helpers below,
// then `await deck.save(path)` and RENDER the file (office_render) before delivering.
//
//   import { createDeck } from "<kit>/deck.mjs"
//   const deck = createDeck({ title: "Q3 review", palette: "editorial", fonts: "editorial" })
//   deck.cover({ kicker: "QUARTERLY REVIEW", title: "Growth held, margin did not", subtitle: "Q3 2026 · Finance" })
//   deck.content({ title: "What changed", bullets: ["Revenue +12% on enterprise renewals", "Gross margin down 2.1 pts"] })
//   await deck.save("out/deck.pptx")
import PptxGenJS from "pptxgenjs"
import { fontPair, hex, palette, scale, series, shade } from "./design.mjs"

const W = 13.333
const H = 7.5
const M = 0.75 // outer margin
const COL = (W - 2 * M) / 12

/** x of grid column n (0-based) and width of k columns */
const cx = (n) => M + n * COL
const cw = (k) => k * COL

export function createDeck(options = {}) {
  const p = palette(options.palette ?? "graphite")
  const f = fontPair(options.fonts ?? "modern")
  const pptx = new PptxGenJS()
  pptx.layout = "LAYOUT_WIDE"
  pptx.title = options.title ?? "Presentation"
  pptx.author = options.author ?? ""
  pptx.theme = { headFontFace: f.head, bodyFontFace: f.body }

  pptx.defineSlideMaster({
    title: "BASE",
    background: { color: hex(p.bg) },
    slideNumber: {
      x: W - M - 0.6,
      y: H - 0.5,
      w: 0.6,
      h: 0.3,
      fontFace: f.body,
      fontSize: scale.caption,
      color: hex(p.muted),
      align: "right",
    },
  })

  let count = 0
  const footer = (slide) => {
    if (!options.footer) return
    slide.addText(options.footer, {
      x: M,
      y: H - 0.5,
      w: 6,
      h: 0.3,
      fontFace: f.body,
      fontSize: scale.caption,
      color: hex(p.muted),
      margin: 0,
    })
  }
  const next = () => {
    count += 1
    return pptx.addSlide({ masterName: "BASE" })
  }
  const title = (slide, text, y = 0.62) => {
    slide.addText(text, {
      x: M,
      y,
      w: W - 2 * M,
      h: 1.0,
      fontFace: f.head,
      fontSize: scale.h1 - 2,
      color: hex(p.ink),
      margin: 0,
      valign: "top",
      fit: "shrink",
    })
    slide.addShape(pptx.ShapeType.line, { x: M, y: y + 1.0, w: 0.9, h: 0, line: { color: hex(p.accent), width: 2.25 } })
  }
  const bullets = (items, o = {}) =>
    items.map((item, index) => ({
      text: item,
      options: {
        bullet: { indent: 16, characterCode: "2013" },
        breakLine: index < items.length - 1,
        paraSpaceAfter: o.gap ?? 16,
        fontSize: o.size ?? scale.lead + 5,
        color: hex(p.ink),
      },
    }))

  const api = {
    pptx,
    palette: p,
    fonts: f,
    grid: { W, H, M, COL, cx, cw },

    cover({ kicker, title: text, subtitle, meta }) {
      const slide = next()
      slide.addShape(pptx.ShapeType.rect, {
        x: 0,
        y: 0,
        w: 0.28,
        h: H,
        fill: { color: hex(p.accent) },
        line: { color: hex(p.accent), width: 0 },
      })
      if (kicker)
        slide.addText(kicker, {
          x: M + 0.2,
          y: 1.0,
          w: 9,
          h: 0.4,
          fontFace: f.body,
          fontSize: scale.small,
          bold: true,
          charSpacing: 4,
          color: hex(p.accent),
          margin: 0,
        })
      slide.addText(text, {
        x: M + 0.2,
        y: 2.0,
        w: 10.4,
        h: 2.6,
        fontFace: f.head,
        fontSize: scale.display,
        color: hex(p.ink),
        margin: 0,
        valign: "top",
        lineSpacingMultiple: 0.95,
        fit: "shrink",
      })
      if (subtitle)
        slide.addText(subtitle, {
          x: M + 0.2,
          y: 4.85,
          w: 9.2,
          h: 0.9,
          fontFace: f.body,
          fontSize: scale.h3,
          color: hex(p.muted),
          margin: 0,
          valign: "top",
        })
      if (meta)
        slide.addText(meta, {
          x: M + 0.2,
          y: H - 1.0,
          w: 9,
          h: 0.4,
          fontFace: f.body,
          fontSize: scale.small,
          color: hex(p.muted),
          margin: 0,
        })
      return slide
    },

    section({ number, title: text, caption }) {
      const slide = next()
      slide.background = { color: hex(p.surface) }
      if (number !== undefined)
        slide.addText(String(number).padStart(2, "0"), {
          x: M,
          y: 1.6,
          w: 3,
          h: 1.6,
          fontFace: f.head,
          fontSize: scale.hero * 1.5,
          color: hex(p.accent),
          margin: 0,
          valign: "top",
        })
      slide.addText(text, {
        x: M,
        y: 3.6,
        w: 10.5,
        h: 1.6,
        fontFace: f.head,
        fontSize: scale.h1 + 4,
        color: hex(p.ink),
        margin: 0,
        valign: "top",
        fit: "shrink",
      })
      if (caption)
        slide.addText(caption, {
          x: M,
          y: 5.3,
          w: 9,
          h: 0.8,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.muted),
          margin: 0,
          valign: "top",
        })
      return slide
    },

    content({ title: heading, bullets: items = [], paragraph, aside, notes }) {
      const slide = next()
      title(slide, heading)
      const wide = aside ? 7 : 10
      if (paragraph)
        slide.addText(paragraph, {
          x: M,
          y: 2.05,
          w: cw(wide),
          h: 4.5,
          fontFace: f.body,
          fontSize: scale.lead + 5,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.25,
        })
      if (items.length)
        slide.addText(bullets(items), {
          x: M,
          y: 2.05,
          w: cw(wide),
          h: 4.5,
          fontFace: f.body,
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.1,
        })
      if (aside) {
        // A pull-out note: accent bar and serif text, sized to its words rather than a tinted box.
        slide.addShape(pptx.ShapeType.line, { x: cx(8.2), y: 2.1, w: 0, h: 1.4, line: { color: hex(p.accent), width: 2.25 } })
        slide.addText(aside, {
          x: cx(8.2) + 0.3,
          y: 2.05,
          w: cw(3.8) - 0.3,
          h: 2.4,
          fontFace: f.head,
          fontSize: scale.h3 + 2,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.2,
        })
      }
      footer(slide)
      if (notes) slide.addNotes(notes)
      return slide
    },

    statement({ text, source }) {
      const slide = next()
      slide.addText(text, {
        x: M,
        y: 1.6,
        w: 10.8,
        h: 3.9,
        fontFace: f.head,
        fontSize: scale.h1 + 4,
        color: hex(p.ink),
        margin: 0,
        valign: "middle",
        lineSpacingMultiple: 1.05,
        fit: "shrink",
      })
      slide.addShape(pptx.ShapeType.line, { x: M, y: 5.75, w: 0.9, h: 0, line: { color: hex(p.accent), width: 2.25 } })
      if (source)
        slide.addText(source, {
          x: M,
          y: 5.95,
          w: 9,
          h: 0.5,
          fontFace: f.body,
          fontSize: scale.small,
          color: hex(p.muted),
          margin: 0,
        })
      return slide
    },

    // One number that carries the slide: value, what it is, and one line of context.
    bigNumber({ title: heading, value, label, caption }) {
      const slide = next()
      if (heading) title(slide, heading)
      slide.addText(String(value), {
        x: M,
        y: 2.0,
        w: 8,
        h: 2.4,
        fontFace: f.head,
        fontSize: scale.hero * 2,
        color: hex(p.accent),
        margin: 0,
        valign: "middle",
        fit: "shrink",
      })
      slide.addText(label, {
        x: M,
        y: 4.5,
        w: 8.5,
        h: 0.7,
        fontFace: f.body,
        fontSize: scale.h3,
        bold: true,
        color: hex(p.ink),
        margin: 0,
        valign: "top",
      })
      if (caption)
        slide.addText(caption, {
          x: M,
          y: 5.2,
          w: 8.5,
          h: 1.1,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.muted),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.15,
        })
      return slide
    },

    // 2-4 numbers side by side: [{ value, label }]
    stats({ title: heading, items }) {
      const slide = next()
      title(slide, heading)
      const n = items.length
      const width = (W - 2 * M - (n - 1) * 0.4) / n
      items.forEach((item, index) => {
        const x = M + index * (width + 0.4)
        slide.addShape(pptx.ShapeType.line, { x, y: 2.8, w: width, h: 0, line: { color: hex(p.rule), width: 1 } })
        slide.addText(String(item.value), {
          x,
          y: 3.0,
          w: width,
          h: 1.5,
          fontFace: f.head,
          fontSize: scale.display,
          color: hex(p.accent),
          margin: 0,
          valign: "top",
          fit: "shrink",
        })
        slide.addText(item.label, {
          x,
          y: 4.6,
          w: width,
          h: 1.4,
          fontFace: f.body,
          fontSize: scale.lead + 3,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.15,
        })
      })
      return slide
    },

    twoColumn({ title: heading, left, right }) {
      const slide = next()
      title(slide, heading)
      const col = (spec, x) => {
        if (spec.heading)
          slide.addText(spec.heading, {
            x,
            y: 2.1,
            w: cw(5.4),
            h: 0.6,
            fontFace: f.head,
            fontSize: scale.h3 + 2,
            color: hex(p.accent),
            margin: 0,
          })
        if (spec.bullets)
          slide.addText(bullets(spec.bullets, { size: scale.lead + 3 }), {
            x,
            y: 2.95,
            w: cw(5.4),
            h: 3.9,
            fontFace: f.body,
            margin: 0,
            valign: "top",
            lineSpacingMultiple: 1.1,
          })
        if (spec.text)
          slide.addText(spec.text, {
            x,
            y: 2.95,
            w: cw(5.4),
            h: 3.9,
            fontFace: f.body,
            fontSize: scale.lead + 3,
            color: hex(p.ink),
            margin: 0,
            valign: "top",
            lineSpacingMultiple: 1.2,
          })
      }
      slide.addShape(pptx.ShapeType.line, { x: cx(6.0), y: 2.15, w: 0, h: 3.9, line: { color: hex(p.rule), width: 1 } })
      col(left, cx(0))
      col(right, cx(6.6))
      return slide
    },

    // Picture on one side, text on the other. `image` is a path or { data: "image/png;base64,..." }.
    imageText({ title: heading, text, bullets: items, image, side = "right" }) {
      const slide = next()
      const imageX = side === "right" ? cx(6.4) : M
      const textX = side === "right" ? M : cx(5.6)
      title(slide, heading)
      const picture = typeof image === "string" ? { path: image } : image
      slide.addImage({ ...picture, x: imageX, y: 2.05, w: cw(5.6), h: 4.4, sizing: { type: "cover", w: cw(5.6), h: 4.4 } })
      if (text)
        slide.addText(text, {
          x: textX,
          y: 2.05,
          w: cw(5.4),
          h: 4.4,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.2,
        })
      if (items)
        slide.addText(bullets(items, { size: scale.lead - 1 }), {
          x: textX,
          y: 2.05,
          w: cw(5.4),
          h: 4.4,
          fontFace: f.body,
          margin: 0,
          valign: "top",
        })
      return slide
    },

    // One step of a user guide: step number, title, a screenshot (whole, never cropped) and what to do.
    // `image` is a PNG path from web_browser; `actions` are short imperatives; `tip` is an optional note.
    step({ number, title: heading, image, actions = [], tip }) {
      const slide = next()
      slide.addShape(pptx.ShapeType.ellipse, { x: M, y: 0.62, w: 0.62, h: 0.62, fill: { color: hex(p.accent) }, line: { color: hex(p.accent) } })
      slide.addText(String(number), { x: M, y: 0.62, w: 0.62, h: 0.62, align: "center", valign: "middle", fontFace: f.head, fontSize: 18, bold: true, color: hex(p.surface), margin: 0 })
      slide.addText(heading, { x: M + 0.85, y: 0.55, w: W - 2 * M - 0.85, h: 0.8, fontFace: f.head, fontSize: scale.h2 ?? 26, bold: true, color: hex(p.ink), margin: 0, valign: "middle", fit: "shrink" })
      const picture = typeof image === "string" ? { path: image } : image
      slide.addShape(pptx.ShapeType.rect, { x: M, y: 1.6, w: cw(8.3), h: 5.1, fill: { color: hex(p.surface) }, line: { color: hex(p.rule), width: 1 } })
      if (picture) slide.addImage({ ...picture, x: M + 0.08, y: 1.68, w: cw(8.3) - 0.16, h: 4.94, sizing: { type: "contain", w: cw(8.3) - 0.16, h: 4.94 } })
      const sideX = cx(8.75)
      const sideW = W - M - sideX
      const numbered = actions.map((item, index) => ({ text: item, options: { bullet: { type: "number", indent: 18 }, breakLine: index < actions.length - 1, paraSpaceAfter: 12, fontSize: scale.lead - 2, color: hex(p.ink) } }))
      slide.addText(numbered, { x: sideX, y: 1.6, w: sideW, h: tip ? 3.7 : 5.1, fontFace: f.body, margin: 0, valign: "top", fit: "shrink" })
      if (tip) {
        slide.addShape(pptx.ShapeType.rect, { x: sideX, y: 5.6, w: sideW, h: 1.1, fill: { color: hex(p.surface) }, line: { color: hex(p.accent), width: 1.25 } })
        slide.addText(tip, { x: sideX + 0.15, y: 5.65, w: sideW - 0.3, h: 1.0, fontFace: f.body, fontSize: scale.small ?? 12, color: hex(p.ink), margin: 0, valign: "middle", fit: "shrink" })
      }
      return slide
    },

    // rows: array of arrays of strings; the first row is the header.
    table({ title: heading, rows, widths, note }) {
      const slide = next()
      title(slide, heading)
      const body = rows.map((row, r) =>
        row.map((cell) => ({
          text: String(cell),
          options: {
            fontFace: f.body,
            fontSize: r === 0 ? scale.small + 1 : scale.lead,
            bold: r === 0,
            color: hex(r === 0 ? p.muted : p.ink),
            align: "left",
            valign: "middle",
            margin: [0.08, 0.1, 0.08, 0.1],
            border: [
              { type: "none" },
              { type: "none" },
              { type: "solid", color: hex(r === 0 ? p.ink : p.rule), pt: r === 0 ? 1.25 : 0.75 },
              { type: "none" },
            ],
          },
        })),
      )
      // `widths` are relative weights; turn them into inches that fill the slide.
      const weight = widths ?? rows[0].map(() => 1)
      const sum = weight.reduce((a, b) => a + b, 0)
      slide.addTable(body, {
        x: M,
        y: 2.15,
        w: W - 2 * M,
        colW: weight.map((value) => ((W - 2 * M) * value) / sum),
        rowH: 0.62,
      })
      if (note)
        slide.addText(note, {
          x: M,
          y: H - 1.0,
          w: W - 2 * M,
          h: 0.4,
          fontFace: f.body,
          fontSize: scale.small,
          color: hex(p.muted),
          margin: 0,
        })
      return slide
    },

    // type: "bar" | "line" | "pie" | "doughnut" | "area". data: [{ name, labels, values }]
    chart({ title: heading, type = "bar", data, takeaway }) {
      const slide = next()
      title(slide, heading)
      const kind = {
        bar: pptx.charts.BAR,
        line: pptx.charts.LINE,
        pie: pptx.charts.PIE,
        doughnut: pptx.charts.DOUGHNUT,
        area: pptx.charts.AREA,
      }[type]
      const radial = type === "pie" || type === "doughnut"
      slide.addChart(kind, data, {
        x: M,
        y: 2.0,
        w: takeaway ? cw(8) : W - 2 * M,
        h: 4.7,
        chartColors: series(p).slice(0, radial ? (data[0]?.values?.length ?? 6) : data.length),
        catAxisLabelFontFace: f.body,
        valAxisLabelFontFace: f.body,
        catAxisLabelFontSize: scale.small,
        valAxisLabelFontSize: scale.small,
        catAxisLabelColor: hex(p.muted),
        valAxisLabelColor: hex(p.muted),
        valGridLine: { color: hex(p.rule), size: 0.5 },
        catGridLine: { style: "none" },
        showLegend: data.length > 1 || radial,
        legendPos: "b",
        legendFontFace: f.body,
        legendFontSize: scale.small,
        legendColor: hex(p.muted),
        barGapWidthPct: 60,
        lineSize: 2.5,
        lineDataSymbolSize: 7,
        dataLabelFontFace: f.body,
        dataLabelFontSize: scale.small,
        dataLabelColor: hex(p.ink),
      })
      if (takeaway) {
        slide.addShape(pptx.ShapeType.line, { x: cx(8.6), y: 2.1, w: 0, h: 4.4, line: { color: hex(p.rule), width: 1 } })
        slide.addText(takeaway, {
          x: cx(9),
          y: 2.1,
          w: cw(3),
          h: 4.4,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.2,
        })
      }
      return slide
    },

    quote({ text, by }) {
      const slide = next()
      slide.addText("“", {
        x: M,
        y: 0.9,
        w: 2,
        h: 1.6,
        fontFace: f.head,
        fontSize: 120,
        color: hex(p.accent),
        margin: 0,
        valign: "top",
      })
      slide.addText(text, {
        x: M + 0.2,
        y: 2.3,
        w: 10.6,
        h: 3.2,
        fontFace: f.head,
        fontSize: scale.h1,
        color: hex(p.ink),
        margin: 0,
        valign: "top",
        lineSpacingMultiple: 1.1,
        fit: "shrink",
      })
      if (by)
        slide.addText(by, {
          x: M + 0.2,
          y: 5.8,
          w: 9,
          h: 0.5,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.muted),
          margin: 0,
        })
      return slide
    },

    // Steps or phases in a row: [{ label, text }]
    timeline({ title: heading, steps }) {
      const slide = next()
      title(slide, heading)
      const width = (W - 2 * M) / steps.length
      slide.addShape(pptx.ShapeType.line, { x: M, y: 3.1, w: W - 2 * M, h: 0, line: { color: hex(p.rule), width: 1.5 } })
      steps.forEach((step, index) => {
        const x = M + index * width
        slide.addShape(pptx.ShapeType.ellipse, {
          x,
          y: 2.97,
          w: 0.26,
          h: 0.26,
          fill: { color: hex(p.accent) },
          line: { color: hex(p.accent), width: 0 },
        })
        slide.addText(step.label, {
          x,
          y: 3.5,
          w: width - 0.3,
          h: 0.6,
          fontFace: f.head,
          fontSize: scale.h3 + 4,
          color: hex(p.ink),
          margin: 0,
          valign: "top",
        })
        slide.addText(step.text, {
          x,
          y: 4.2,
          w: width - 0.3,
          h: 2.2,
          fontFace: f.body,
          fontSize: scale.lead + 1,
          color: hex(p.muted),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.2,
        })
      })
      return slide
    },

    closing({ title: text, lines = [] }) {
      const slide = next()
      slide.background = { color: hex(p.surface) }
      slide.addText(text, {
        x: M,
        y: 2.2,
        w: 10.6,
        h: 1.8,
        fontFace: f.head,
        fontSize: scale.display,
        color: hex(p.ink),
        margin: 0,
        valign: "top",
        fit: "shrink",
      })
      slide.addShape(pptx.ShapeType.line, { x: M, y: 4.2, w: 0.9, h: 0, line: { color: hex(p.accent), width: 2.25 } })
      if (lines.length)
        slide.addText(lines.join("\n"), {
          x: M,
          y: 4.5,
          w: 9,
          h: 1.6,
          fontFace: f.body,
          fontSize: scale.lead,
          color: hex(p.muted),
          margin: 0,
          valign: "top",
          lineSpacingMultiple: 1.3,
        })
      return slide
    },

    async save(file) {
      await pptx.writeFile({ fileName: file })
      return { file, slides: count }
    },
  }
  return api
}

export { shade }
