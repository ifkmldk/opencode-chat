// OpenCode office kit: rasterize PDF pages to PNG with MuPDF (WASM, no native modules), run with node.
//   node render-pdf.mjs <pdf> <outDir> [scale=1.5] [pages=all|1,3-5] [prefix=page]
// Prints one JSON object: { pages: number, files: [{ page, file, width, height }] }.
import fs from "node:fs"
import path from "node:path"
import * as mupdf from "mupdf"

const [, , input, outDir, scaleArg = "1.5", pagesArg = "all", prefix = "page"] = process.argv
const scale = Math.max(0.5, Math.min(4, Number(scaleArg) || 1.5))

function pick(total, spec) {
  if (spec === "all") return Array.from({ length: total }, (_, index) => index + 1)
  const wanted = new Set()
  for (const part of spec.split(",")) {
    const [from, to = from] = part.split("-").map((value) => Number.parseInt(value, 10))
    if (!Number.isFinite(from) || !Number.isFinite(to)) continue
    for (let page = Math.max(1, from); page <= Math.min(total, to); page++) wanted.add(page)
  }
  return [...wanted].sort((a, b) => a - b)
}

const doc = mupdf.Document.openDocument(fs.readFileSync(input), "application/pdf")
const total = doc.countPages()
fs.mkdirSync(outDir, { recursive: true })
const files = []
for (const number of pick(total, pagesArg)) {
  const page = doc.loadPage(number - 1)
  const pixmap = page.toPixmap(mupdf.Matrix.scale(scale, scale), mupdf.ColorSpace.DeviceRGB, false, true)
  const file = path.join(outDir, `${prefix}-${String(number).padStart(2, "0")}.png`)
  fs.writeFileSync(file, pixmap.asPNG())
  files.push({ page: number, file, width: pixmap.getWidth(), height: pixmap.getHeight() })
}
process.stdout.write(JSON.stringify({ pages: total, files }))
