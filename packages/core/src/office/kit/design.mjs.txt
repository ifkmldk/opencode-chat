// Modular type scale (1.25) in points, anchored at body 12.
export const scale = { caption: 9, small: 10.5, body: 12, lead: 15, h3: 18, h2: 24, h1: 32, display: 48, hero: 64 }

export function palette(name = "graphite") {
  const found = palettes[name]
  if (!found) throw new Error(`Unknown palette "${name}". Choose one of: ${Object.keys(palettes).join(", ")}`)
  return found
}

export function fontPair(name = "modern") {
  const found = fonts[name]
  if (!found) throw new Error(`Unknown font pair "${name}". Choose one of: ${Object.keys(fonts).join(", ")}`)
  return found
}

/** "#B5532A" or "B5532A" -> "B5532A" (office formats want bare hex). */
export const hex = (value) => String(value).replace(/^#/, "").toUpperCase()

/** Lighten (amount > 0) or darken (amount < 0) a bare hex colour; amount is -1..1. */
export function shade(value, amount) {
  const color = hex(value)
  const channel = (index) => {
    const base = Number.parseInt(color.slice(index, index + 2), 16)
    const mixed = amount >= 0 ? base + (255 - base) * amount : base * (1 + amount)
    return Math.max(0, Math.min(255, Math.round(mixed))).toString(16).padStart(2, "0")
  }
  return `${channel(0)}${channel(2)}${channel(4)}`.toUpperCase()
}

/** A categorical series for charts: the accent first, then quiet companions; never more than six. */
export function series(p) {
  return [p.accent, p.accent2, shade(p.accent, 0.45), shade(p.accent2, 0.45), p.muted, shade(p.muted, 0.5)]
}
